table users {
  id string PK // uuid
  name string // nullable
  phone_e164 string
  phone_verified_at datetime // nullable
  email string // nullable
  email_verified_at datetime // nullable
  password_hash string
  role UserRole
  status AccountStatus
  preferences jsonb
  created_at datetime
  updated_at datetime
}

  created_at datetime
  updated_at datetime
}

// Admin seeds canonical ingredient data.
table master_ingredients {
  id string PK // uuid
  name string
  description string
  category_id string  // ingredient_categories.id
  default_media_url string // nullable
  canonical_unit MeasurementUnit
  calories float // nullable, per 100 g where applicable
  protein_g float // nullable
  fat_g float // nullable
  carbs_g float // nullable
  sugar_g float // nullable
  sodium_mg float // nullable
  other_nutrients jsonb
  default_storage_mode StorageMode // nullable
  created_at datetime
  updated_at datetime
}
// Hỗ trợ cho chức năng tìm kiếm. Ví dụ: thịt lợn -> thịt heo
table ingredient_aliases {
  id string PK // uuid
  master_ingredient_id string  // master_ingredients.id
  alias string
  normalized_alias string
  created_at datetime
}

table shelf_life_rules {
  id string PK // uuid
  scope ShelfLifeRuleScope

  category_id string  // nullable, ingredient_categories.id
  storage_mode StorageMode
  min_days integer
  max_days integer
  default_days integer
  created_at datetime
  updated_at datetime
}

  total_fat_g float // nullable
  total_carbs_g float // nullable
  total_sugar_g float // nullable

  updated_at datetime

// Each row is a separate user-owned batch; never merge different expiry dates.
table inventory_batches {
  id string PK // uuid
  user_id string  // users.id
  master_ingredient_id string  // nullable, master_ingredients.id
  custom_name string // nullable
  batch_type InventoryBatchType
  initial_quantity float
  current_quantity float
  unit MeasurementUnit
  storage_mode StorageMode
  status InventoryBatchStatus
  purchased_at datetime // nullable
  packaged_at datetime // nullable
  stored_at datetime // nullable
  expires_at datetime // nullable
  expiration_source ExpirationSource
  unit_cost float // nullable
  note string // nullable
  media_url string // nullable
  source InventorySource
  source_cooking_session_id string  // nullable, cooking_sessions.id
  created_at datetime
  updated_at datetime
  archived_at datetime // nullable
}

table inventory_ledger_entries {
  id string PK // uuid
  user_id string  // users.id

}
